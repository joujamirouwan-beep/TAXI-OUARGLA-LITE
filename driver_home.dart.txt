import 'dart:async';

import 'package:flutter/material.dart';
import 'package:geolocator/geolocator.dart';
import 'package:google_maps_flutter/google_maps_flutter.dart';

import '../services/trip_service.dart';

class DriverHome extends StatefulWidget {
  const DriverHome({super.key});

  @override
  State<DriverHome> createState() => _DriverHomeState();
}

class _DriverHomeState extends State<DriverHome> {
  final TripService _trips = TripService();

  static const LatLng _ouargla = LatLng(31.9522, 5.3242);

  LatLng _position = _ouargla;
  Map<String, dynamic>? _activeRide;
  List<Map<String, dynamic>> _requests = [];

  StreamSubscription<List<Map<String, dynamic>>>? _requestsSub;
  StreamSubscription<Position>? _positionSub;

  bool _online = false;
  bool _loading = true;
  bool _busy = false;

  @override
  void initState() {
    super.initState();
    _initialize();
  }

  Future<void> _initialize() async {
    try {
      final values = await Future.wait([
        _trips.getDriverOnlineStatus(),
        _trips.getActiveDriverRide(),
      ]);
      if (!mounted) return;

      setState(() {
        _online = values[0] as bool;
        _activeRide = values[1] as Map<String, dynamic>?;
      });

      _listenForRequests();
      if (_activeRide != null) await _startLocationTracking();
    } catch (e) {
      _message('تعذر تحميل حساب السائق: $e');
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  void _listenForRequests() {
    _requestsSub?.cancel();
    _requestsSub = _trips.streamPendingTrips().listen(
      (rows) {
        if (!mounted) return;
        setState(() => _requests = rows);
      },
      onError: (Object error) => _message('خطأ في استقبال الطلبات: $error'),
    );
  }

  Future<void> _setOnline(bool value) async {
    if (_busy) return;
    setState(() => _busy = true);
    try {
      final ok = await _trips.setDriverOnline(value);
      if (!ok) throw Exception('لم يتم تغيير حالة الاتصال');
      if (!mounted) return;
      setState(() => _online = value);
      _message(value ? 'أنت متصل الآن' : 'أنت غير متصل الآن');
    } catch (e) {
      _message('تعذر تغيير الاتصال: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<Position> _currentPosition() async {
    if (!await Geolocator.isLocationServiceEnabled()) {
      throw Exception('يرجى تفعيل GPS');
    }

    var permission = await Geolocator.checkPermission();
    if (permission == LocationPermission.denied) {
      permission = await Geolocator.requestPermission();
    }
    if (permission == LocationPermission.denied ||
        permission == LocationPermission.deniedForever) {
      throw Exception('إذن الموقع غير متاح');
    }

    return Geolocator.getCurrentPosition(
      locationSettings: const LocationSettings(
        accuracy: LocationAccuracy.high,
        timeLimit: Duration(seconds: 20),
      ),
    );
  }

  Future<void> _startLocationTracking() async {
    if (_activeRide == null) return;
    await _positionSub?.cancel();

    try {
      // Request/check location permission before opening the live stream.
      final initialPosition = await _currentPosition();
      if (!mounted || _activeRide == null) return;
      setState(() => _position = LatLng(
            initialPosition.latitude,
            initialPosition.longitude,
          ));
      await _publishPosition(initialPosition);

      _positionSub = Geolocator.getPositionStream(
        locationSettings: const LocationSettings(
          accuracy: LocationAccuracy.high,
          distanceFilter: 15,
        ),
      ).listen((position) {
        if (!mounted || _activeRide == null) return;
        setState(() => _position = LatLng(
              position.latitude,
              position.longitude,
            ));
        _publishPosition(position);
      }, onError: (Object error) {
        debugPrint('GPS stream error: $error');
        _message('توقف تحديث الموقع: $error');
      });
    } catch (error) {
      _message('تعذر تشغيل تتبع الموقع: $error');
    }
  }

  Future<void> _publishPosition(Position p) async {
    final ride = _activeRide;
    if (ride == null) return;
    try {
      await _trips.updateDriverLocation(
        rideId: ride['id'].toString(),
        latitude: p.latitude,
        longitude: p.longitude,
      );
    } catch (e) {
      debugPrint('Location update failed: $e');
    }
  }

  void _stopLocationTracking() {
    _positionSub?.cancel();
    _positionSub = null;
  }

  Future<void> _refreshActiveRide() async {
    final ride = await _trips.getActiveDriverRide();
    if (!mounted) return;
    setState(() => _activeRide = ride);
    if (ride == null) {
      _stopLocationTracking();
    } else {
      await _startLocationTracking();
    }
  }

  Future<void> _accept(String rideId) async {
    if (_busy || _activeRide != null) return;
    setState(() => _busy = true);
    try {
      final ok = await _trips.acceptTrip(rideId);
      if (!ok) throw Exception('الطلب لم يعد متاحاً');
      await _refreshActiveRide();
      _message('تم قبول الرحلة');
    } catch (e) {
      _message('تعذر قبول الرحلة: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  String _nextStatus(String status) {
    switch (status) {
      case 'accepted':
        return 'arrived';
      case 'arrived':
        return 'in_progress';
      case 'in_progress':
        return 'completed';
      default:
        return '';
    }
  }

  String _statusText(String status) {
    switch (status) {
      case 'accepted':
        return 'تم قبول الرحلة';
      case 'arrived':
        return 'وصلت إلى الراكب';
      case 'in_progress':
        return 'الرحلة جارية';
      default:
        return status;
    }
  }

  String _actionText(String status) {
    switch (status) {
      case 'accepted':
        return 'وصلت إلى مكان الراكب';
      case 'arrived':
        return 'بدء الرحلة';
      case 'in_progress':
        return 'إنهاء الرحلة';
      default:
        return 'تحديث الحالة';
    }
  }

  Future<void> _advanceRide() async {
    final ride = _activeRide;
    if (ride == null || _busy) return;
    final next = _nextStatus(ride['status'].toString());
    if (next.isEmpty) return;

    setState(() => _busy = true);
    try {
      final ok = await _trips.setRideStatus(
        rideId: ride['id'].toString(),
        nextStatus: next,
      );
      if (!ok) throw Exception('لم يتم تحديث حالة الرحلة');
      await _refreshActiveRide();
      _message(next == 'completed' ? 'اكتملت الرحلة' : 'تم تحديث الحالة');
    } catch (e) {
      _message('تعذر تحديث الرحلة: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  void _message(String value) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(value, textAlign: TextAlign.right)),
    );
  }

  @override
  void dispose() {
    _requestsSub?.cancel();
    _stopLocationTracking();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(
          title: const Text('طاكسي ورڨلة لايت - السائق'),
          actions: [
            Center(child: Text(_online ? 'متصل' : 'غير متصل')),
            Switch(
              value: _online,
              onChanged: _busy ? null : _setOnline,
            ),
            const SizedBox(width: 8),
          ],
        ),
        body: _loading
            ? const Center(child: CircularProgressIndicator())
            : Stack(
                children: [
                  GoogleMap(
                    initialCameraPosition: const CameraPosition(
                      target: _ouargla,
                      zoom: 14,
                    ),
                    myLocationEnabled: false,
                    markers: {
                      Marker(
                        markerId: const MarkerId('driver'),
                        position: _position,
                        infoWindow: const InfoWindow(title: 'موقع السائق'),
                      ),
                    },
                  ),
                  Align(
                    alignment: Alignment.bottomCenter,
                    child: _panel(),
                  ),
                ],
              ),
      ),
    );
  }

  Widget _panel() {
    return Container(
      width: double.infinity,
      constraints: const BoxConstraints(maxHeight: 340),
      padding: const EdgeInsets.all(16),
      decoration: const BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.vertical(top: Radius.circular(20)),
      ),
      child: SingleChildScrollView(
        child: _activeRide != null
            ? _activeRideCard()
            : _online
                ? _requestList()
                : const Padding(
                    padding: EdgeInsets.all(16),
                    child: Text(
                      'أنت غير متصل. فعّل الاتصال لاستقبال الطلبات.',
                      textAlign: TextAlign.center,
                    ),
                  ),
      ),
    );
  }

  Widget _activeRideCard() {
    final ride = _activeRide!;
    final status = ride['status'].toString();
    final next = _nextStatus(status);

    return Column(
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [
        const Text(
          'الرحلة النشطة',
          style: TextStyle(fontSize: 19, fontWeight: FontWeight.bold),
        ),
        const SizedBox(height: 10),
        Text('الحالة: ${_statusText(status)}'),
        Text('من: ${ride['pickup_address'] ?? ''}'),
        Text('إلى: ${ride['dropoff_address'] ?? ''}'),
        const SizedBox(height: 12),
        if (next.isNotEmpty)
          ElevatedButton(
            onPressed: _busy ? null : _advanceRide,
            child: _busy
                ? const SizedBox(
                    height: 20,
                    width: 20,
                    child: CircularProgressIndicator(strokeWidth: 2),
                  )
                : Text(_actionText(status)),
          ),
      ],
    );
  }

  Widget _requestList() {
    if (_requests.isEmpty) {
      return const Padding(
        padding: EdgeInsets.all(18),
        child: Text('لا توجد طلبات جديدة حالياً', textAlign: TextAlign.center),
      );
    }

    return Column(
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [
        const Text(
          'طلبات الرحلات المتاحة',
          style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
        ),
        ..._requests.map((ride) => Card(
              child: ListTile(
                title: Text('من: ${ride['pickup_address'] ?? ''}'),
                subtitle: Text('إلى: ${ride['dropoff_address'] ?? ''}'),
                trailing: ElevatedButton(
                  onPressed: _busy || _activeRide != null
                      ? null
                      : () => _accept(ride['id'].toString()),
                  child: const Text('قبول'),
                ),
              ),
            )),
      ],
    );
  }
}
